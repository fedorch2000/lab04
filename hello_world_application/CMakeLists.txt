cmake_minimum_required(VERSION 3.4)
project(hello_world_application)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(formatter STATIC ${CMAKE_CURRENT_SOURCE_DIR}/../formatter_lib/formatter.cpp ${CMAKE_CURRENT_SOURCE_DIR}/../formatter_lib/formatter.h)
add_library(formatter_ex STATIC ${CMAKE_CURRENT_SOURCE_DIR}/../formatter_ex_lib/formatter_ex.cpp ${CMAKE_CURRENT_SOURCE_DIR}/../formatter_ex_lib/formatter_ex.h)
target_link_libraries(formatter_ex formatter)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/../formatter_lib/)
add_executable(hello_world_application ${CMAKE_CURRENT_SOURCE_DIR}/hello_world.cpp)
target_link_libraries(hello_world_application formatter_ex)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/../formatter_ex_lib/)
